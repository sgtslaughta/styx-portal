# UI Polish Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** One top nav, one master–detail home (unified workspace list + a detail pane for both workstations and instances), then Settings, shell and global polish.

**Architecture:** `App.tsx` owns a single `MainTab` (`workspaces | apps | templates | system`); `HomeTabs` is deleted. The Workspaces tab is `InstanceWorkspace` = `WorkspaceList` (search + `WorkstationGroup` + `InstanceGroup`, sticky collapsible headers, one `WorkspaceRow` look) on the left and a pane chosen by `Selection {kind, id}` on the right (`WorkstationDetailPane` or the existing `InstanceDetailPane`, both framed by `PaneHeader`). Connect logic moves into two hooks (`useWorkstationConnect`, `useInstanceConnect`) that also write the most-recent workspace to `localStorage`. Pure formatting lives in `lib/workspace.ts`; never-throwing storage in `lib/local.ts`.

**Tech Stack:** React 19, TypeScript, Vite, Tailwind 4.1 (has `pointer-coarse:`), shadcn/ui (radix-ui), TanStack Query 5, framer-motion, lucide-react, vitest 2 + Testing Library, Playwright (e2e, run by the controller).

`frontend/src/lib/local.ts`:

```ts
/** localStorage that never throws (private mode, blocked storage): reads fall back to null, writes are dropped. */
export function readLocal(key: string): string | null {
  try { return localStorage.getItem(key); } catch { return null; }
}

/** `null` removes the key. */
export function writeLocal(key: string, value: string | null): void {
  try {
    if (value === null) localStorage.removeItem(key);
    else localStorage.setItem(key, value);
  } catch { /* storage blocked: session-only */ }
}
```

`frontend/src/lib/workspace.ts`:

```ts
import type { Workstation } from "@/api/client";
import type { Instance } from "@/lib/types";
import { readLocal, writeLocal } from "@/lib/local";
import { isRunning, statusMeta } from "@/lib/status";
import { formatDuration, shortError } from "@/lib/utils";

export type WorkspaceKind = "workstation" | "instance";
export type Selection = { kind: WorkspaceKind; id: string };

const RECENT_KEY = "styx.home.recent";

/** The workspace last opened (Connect/Open) in this browser, or null. */
export function readRecent(): Selection | null {
  try {
    const v = JSON.parse(readLocal(RECENT_KEY) ?? "null") as Partial<Selection> | null;
    if (v && (v.kind === "workstation" || v.kind === "instance") && typeof v.id === "string") return { kind: v.kind, id: v.id };
  } catch { /* garbage: no recent */ }
  return null;
}

export function writeRecent(s: Selection): void {
  writeLocal(RECENT_KEY, JSON.stringify(s));
}

/** Compact age: "4s", "5m", "3h", "1 day", "2 days". Null or unparsable input -> null. */
export function since(iso: string | null, now = Date.now()): string | null {
  if (!iso) return null;
  const t = new Date(iso).getTime();
  if (Number.isNaN(t)) return null;
  const s = Math.max(0, Math.floor((now - t) / 1000));
  if (s < 60) return `${s}s`;
  if (s < 3600) return `${Math.floor(s / 60)}m`;
  if (s < 86_400) return `${Math.floor(s / 3600)}h`;
  const d = Math.floor(s / 86_400);
  return `${d} day${d === 1 ? "" : "s"}`;
}

/** One secondary line per workstation row. */
export function wsLine(ws: Workstation, now = Date.now()): string {
  const seen = since(ws.last_heartbeat, now);
  if (ws.status !== "online") return seen ? `Offline · ${seen}` : "Offline";
  if (ws.in_use && !ws.in_use_self) return `In use by ${ws.in_use_by ?? "someone"}`;
  if (ws.in_use_self) return "Connected (you)";
  const pn = ws.os_info?.pretty_name;
  const os = typeof pn === "string" && pn ? pn : "Linux";
  return seen ? `${os} · seen ${seen} ago` : os;
}

export function wsDot(ws: Workstation): { dotClass: string; label: string } {
  if (ws.status !== "online") return { dotClass: "bg-muted-foreground", label: "Offline" };
  if (ws.last_error) return { dotClass: "bg-destructive", label: "Error" };
  if (ws.in_use) return { dotClass: "bg-warning", label: ws.in_use_self ? "Connected (you)" : `In use by ${ws.in_use_by ?? "someone"}` };
  return { dotClass: "bg-success", label: "Online" };
}

/** One secondary line per instance row: "Running · up 6d 5h" | "Stopped" | "Error · <short reason>". */
export function instLine(inst: Instance, now = Date.now()): string {
  const label = statusMeta(inst.status).label;
  if (inst.started_at && isRunning(inst.status)) {
    return `${label} · up ${formatDuration((now - new Date(inst.started_at + "Z").getTime()) / 1000)}`;
  }
  if (inst.status === "error" && inst.error_message) return `Error · ${shortError(inst.error_message, 40)}`;
  return label;
}
```

```ts
/** Largest two units, minute precision: "6d 5h", "3h 12m", "45m"; seconds only under a minute. */
export function formatDuration(seconds: number | null | undefined): string {
  if (seconds == null || !Number.isFinite(seconds) || seconds < 0) return "—";
  if (seconds < 60) return `${Math.floor(seconds)}s`;
  const s = Math.floor(seconds);
  const d = Math.floor(s / 86_400);
  const h = Math.floor((s % 86_400) / 3600);
  const m = Math.floor((s % 3600) / 60);
  const [big, small] = d ? [`${d}d`, h ? `${h}h` : ""] : h ? [`${h}h`, m ? `${m}m` : ""] : [`${m}m`, ""];
  return small ? `${big} ${small}` : big;
}
```

```tsx
import { readLocal, writeLocal } from "@/lib/local";

export type MainTab = "workspaces" | "apps" | "templates" | "system";

const HOME_KEY = "styx.home.tab"; // kept from the old Workspaces | Apps sub-tabs
const TABS: { id: MainTab; label: string }[] = [
  { id: "workspaces", label: "Workspaces" },
  { id: "apps", label: "Apps" },
  { id: "templates", label: "Templates" },

/** Landing tab: the last of Workspaces/Apps used in this browser. */
export function readHomeTab(): MainTab {
```

```tsx
import { useState } from "react";
import { api, ApiError, type OpenMode, type Workstation } from "@/api/client";
import { ConfirmDialog } from "@/components/common/confirm-dialog";
import { useOpenChoice } from "@/components/open-prompt";
import { openPending } from "@/lib/connect-wipe";
import { writeRecent } from "@/lib/workspace";

type Ask = { ws: Workstation; kind: "join" | "takeover"; mode?: OpenMode };

/** Connect / Reconnect / Join / Take over for workstations. Render `dialog` once next to the buttons.
 *  `mode` skips the open-preference prompt (e.g. "Open in new tab"). */
export function useWorkstationConnect() {
  const choose = useOpenChoice();
  const [ask, setAsk] = useState<Ask | null>(null);
  const [error, setError] = useState<string | null>(null);

  function open(ws: Workstation, force: boolean, mode?: OpenMode) {
    setError(null);
    writeRecent({ kind: "workstation", id: ws.id });
    const run = (m: OpenMode) => {
      openPending(m, `Connecting to ${ws.name}…`, async () => (await api.workstationConnectUrl(ws.id, force)).url)
        .catch((e: unknown) => {
          if (e instanceof ApiError && e.status === 423) setAsk({ ws, kind: "join", mode }); // raced: someone joined meanwhile
          else setError((e as Error).message);
        });
    };
    if (mode) run(mode);
    else choose("workspaces", ws.name, run);
  }

  const connect = (ws: Workstation, mode?: OpenMode) =>
    ws.in_use && !ws.in_use_self ? setAsk({ ws, kind: "join", mode }) : open(ws, false, mode);
  const takeOver = (ws: Workstation) => setAsk({ ws, kind: "takeover" });

  const who = ask?.ws.in_use_by ?? "another user";
  const takeover = ask?.kind === "takeover";
  const dialog = (
    <ConfirmDialog
      open={ask !== null}
      onOpenChange={(o) => { if (!o) setAsk(null); }}
      title={takeover ? `Take over "${ask?.ws.name ?? ""}"?` : `Join "${ask?.ws.name ?? ""}"?`}
      description={takeover
        ? `${who} is using this workstation. You get the seat; their session is not closed, so agree on it first.`
        : `This workstation is in use by ${who}. Joining shares the same desktop — you will both see and control the same screen.`}
      confirmLabel={takeover ? "Take over" : "Join"}
      variant={takeover ? "destructive" : "default"}
      onConfirm={() => {
        if (!ask) return;
        setAsk(null);
        open(ask.ws, true, ask.mode);
      }}
    />
  );
  return { connect, takeOver, dialog, error };
}
```

```tsx
/** Collapsed state of a home group, remembered per browser. */
export function useCollapsed(group: string): [boolean, () => void] {
  const key = `styx.home.collapsed.${group}`;
  const [collapsed, setCollapsed] = useState(() => readLocal(key) === "1");
  const toggle = () => {
    const next = !collapsed;
    setCollapsed(next);
    writeLocal(key, next ? "1" : null);
  };
  return [collapsed, toggle];
}
```

```tsx
import { useMemo, useState } from "react";
import type { Workstation } from "@/api/client";
import { WorkstationIcon } from "@/components/system/workstation-icon";
import { useAuth } from "@/hooks/use-auth";
import { useMyWorkstations } from "@/hooks/use-workstations";
import { useWorkstationConnect } from "@/hooks/use-workstation-connect";
import { wsDot, wsLine, type Selection } from "@/lib/workspace";
import { EmptyLine, GroupHeader, WorkspaceRow, useCollapsed } from "./workspace-row";

type SortKey = "status" | "name" | "seen";
const rank = (ws: Workstation) => (ws.status !== "online" ? 2 : ws.in_use ? 1 : 0);
const LIST_ID = "home-workstations";

export function WorkstationGroup({ query, selection, onSelect, onOpenSettings }: {
  query: string; selection: Selection | null; onSelect: (s: Selection) => void; onOpenSettings?: (section?: string) => void;
}) {
  const { data } = useMyWorkstations();
  const { user } = useAuth();
  const [collapsed, toggle] = useCollapsed("workstations");
  const [sort, setSort] = useState<SortKey>("status");
  const { connect, dialog, error } = useWorkstationConnect();

  const rows = useMemo(() => {
    const q = query.trim().toLowerCase();
    return (data ?? [])
      .filter((ws) => !q || `${ws.name} ${ws.hostname}`.toLowerCase().includes(q))
      .sort((a, b) =>
        sort === "name" ? a.name.localeCompare(b.name)
          : sort === "seen" ? (b.last_heartbeat ?? "").localeCompare(a.last_heartbeat ?? "")
          : rank(a) - rank(b) || a.name.localeCompare(b.name));
  }, [data, query, sort]);

  if (!data) return null;
  const isAdmin = user?.role === "admin";

  return (
    <section aria-label="Workstations">
      <GroupHeader title="Workstations" count={data.length} collapsed={collapsed} onToggle={toggle} controlsId={LIST_ID}>
        <select aria-label="Sort workstations" value={sort} onChange={(e) => setSort(e.target.value as SortKey)}
          className="h-8 rounded-md border border-border bg-background px-1.5 text-xs max-sm:h-11">
          <option value="status">Status</option>
          <option value="name">Name</option>
          <option value="seen">Last seen</option>
        </select>
      </GroupHeader>
      {!collapsed && (
        <div id={LIST_ID} className="py-1">
          {error && <p role="alert" className="px-2 text-xs text-destructive">{error}</p>}
          {data.length === 0 ? (
            <EmptyLine
              text={isAdmin ? "No workstations yet." : "No workstations shared with you yet."}
              action={isAdmin && onOpenSettings ? { label: "Add a workstation", onClick: () => onOpenSettings("workstations") } : undefined}
            />
          ) : rows.length === 0 ? (
            <p className="px-2 py-2 text-xs text-muted-foreground">No workstations match.</p>
          ) : (
            <ul className="space-y-0.5">
              {rows.map((ws) => (
                <WorkspaceRow
                  key={ws.id}
                  icon={<WorkstationIcon ws={ws} className="size-8" />}
                  name={ws.name}
                  line={wsLine(ws)}
                  dot={wsDot(ws)}
                  dim={ws.status !== "online"}
                  selected={selection?.kind === "workstation" && selection.id === ws.id}
                  onSelect={() => onSelect({ kind: "workstation", id: ws.id })}
                  quick={ws.status === "online" ? { label: `Connect to ${ws.name}`, onClick: () => connect(ws) } : undefined}
                />
              ))}
            </ul>
          )}
        </div>
      )}
      {dialog}
    </section>
  );
}
```

```tsx
import { useState } from "react";
import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
import { Pause, Play, Square, Trash2, X } from "lucide-react";
import { toast } from "sonner";
import { Button } from "@/components/ui/button";
import { ConfirmDialog } from "@/components/common/confirm-dialog";
import {
  useDeleteInstance, usePauseInstance, useStartInstance, useStopInstance, useUnpauseInstance,
} from "@/hooks/use-instances";
import { shortError } from "@/lib/utils";
import type { Instance } from "@/lib/types";

type IdMutation = { mutate: (id: string, opts: { onError: (e: Error) => void }) => void };

/** Floating bar acting on the multi-selected instances. */
export function InstanceBulkBar({ selected, onClear }: { selected: Instance[]; onClear: () => void }) {
  const reduce = useReducedMotion();
  const [confirm, setConfirm] = useState(false);
  const start = useStartInstance();
  const stop = useStopInstance();
  const pause = usePauseInstance();
  const unpause = useUnpauseInstance();
  const destroy = useDeleteInstance();
  const busy = start.isPending || stop.isPending || pause.isPending || unpause.isPending || destroy.isPending;
  const running = selected.filter((i) => i.status === "running" || i.status === "idle");
  const stopped = selected.filter((i) => i.status === "stopped" || i.status === "error");
  const paused = selected.filter((i) => i.status === "paused");
  const fail = (i: Instance) => (e: Error) => { console.error(i.name, e); toast.error(shortError(`${i.name}: ${e.message}`)); };

  function bulk(list: Instance[], mut: IdMutation, verb: string) {
    list.forEach((i) => mut.mutate(i.id, { onError: fail(i) }));
    toast.success(`${verb} ${list.length} instance(s)`);
    onClear();
  }
  function destroyAll() {
    selected.forEach((i) => destroy.mutate({ id: i.id, removeVolumes: false }, { onError: fail(i) }));
    toast.success(`Destroying ${selected.length} instance(s)`);
    onClear();
  }

  return (
    <>
      <AnimatePresence>
        {selected.length > 0 && (
          <motion.div
            initial={reduce ? undefined : { y: 60, opacity: 0 }}
            animate={reduce ? undefined : { y: 0, opacity: 1 }}
            exit={reduce ? undefined : { y: 60, opacity: 0 }}
            transition={{ type: "spring", stiffness: 400, damping: 30 }}
            className="fixed bottom-4 left-1/2 z-50 flex -translate-x-1/2 items-center gap-2 rounded-xl border border-border bg-card px-4 py-2.5 shadow-2xl"
          >
            <span className="mr-1 text-xs font-medium text-foreground">{selected.length} selected</span>
            {stopped.length > 0 && (
              <Button size="sm" variant="secondary" disabled={busy} onClick={() => bulk(stopped, start, "Starting")}>
                <Play /> Start {stopped.length}
              </Button>
            )}
            {running.length > 0 && (
              <Button size="sm" variant="secondary" disabled={busy} onClick={() => bulk(running, pause, "Pausing")}>
                <Pause /> Pause {running.length}
              </Button>
            )}
            {paused.length > 0 && (
              <Button size="sm" variant="secondary" disabled={busy} onClick={() => bulk(paused, unpause, "Resuming")}>
                <Play /> Resume {paused.length}
              </Button>
            )}
            {running.length + paused.length > 0 && (
              <Button size="sm" variant="secondary" disabled={busy} onClick={() => bulk([...running, ...paused], stop, "Stopping")}>
                <Square /> Stop {running.length + paused.length}
              </Button>
            )}
            <Button size="sm" variant="ghost" className="text-destructive hover:text-destructive" disabled={busy} onClick={() => setConfirm(true)}>
              <Trash2 /> Destroy
            </Button>
            <div className="mx-1 h-5 w-px bg-border" />
            <Button size="icon-sm" variant="ghost" onClick={onClear} title="Clear selection" aria-label="Clear selection">
              <X />
            </Button>
          </motion.div>
        )}
      </AnimatePresence>
      <ConfirmDialog
        open={confirm}
        onOpenChange={setConfirm}
        title={`Destroy ${selected.length} instance(s)?`}
        description="Containers will be removed. Named volumes are kept."
        confirmLabel="Destroy"
        variant="destructive"
        confirmPhrase="destroy"
        onConfirm={destroyAll}
      />
    </>
  );
}
```

```tsx
import { useInstanceConnect } from "@/hooks/use-instance-connect";
import { isRunning, statusMeta } from "@/lib/status";
import { instLine, type Selection } from "@/lib/workspace";
import { cn } from "@/lib/utils";
import type { Instance } from "@/lib/types";
import { InstanceBulkBar } from "./instance-bulk-bar";
import { EmptyLine, GroupHeader, TemplateIcon, WorkspaceRow, useCollapsed } from "./workspace-row";

type StatusFilter = "all" | "running" | "stopped" | "paused";
const FILTERS: StatusFilter[] = ["all", "running", "stopped", "paused"];
const FILTER_LABELS: Record<StatusFilter, string> = { all: "All", running: "Running", stopped: "Stopped", paused: "Paused" };
const ORDER: Record<string, number> = {
  running: 0, idle: 1, pulling: 2, starting: 3, creating: 4, paused: 5, stopping: 6, stopped: 7, error: 8, created: 9,
};
const COUNTS = [
  { label: "running", dot: "bg-success", match: (s: string) => s === "running" || s === "idle" },
  { label: "paused", dot: "bg-warning", match: (s: string) => s === "paused" },
  { label: "stopped", dot: "bg-muted-foreground", match: (s: string) => s === "stopped" },
  { label: "error", dot: "bg-destructive", match: (s: string) => s === "error" },
];
const LIST_ID = "home-instances";

function byFilter(i: Instance, f: StatusFilter): boolean {
  if (f === "running") return isRunning(i.status);
  if (f === "paused") return i.status === "paused";
  if (f === "stopped") return i.status === "stopped" || i.status === "error";
  return true;
}

export function InstanceGroup({ query, selection, onSelect, onLaunch }: {
  query: string; selection: Selection | null; onSelect: (s: Selection) => void; onLaunch: () => void;
}) {
  const { data: instances, isError, refetch } = useInstances();
  const { data: templates } = useTemplates();
  const connect = useInstanceConnect();
  const [collapsed, toggle] = useCollapsed("instances");
  const [filter, setFilter] = useState<StatusFilter>("all");
  const [picked, setPicked] = useState<Set<string>>(new Set());
  const tmpl = useCallback((i: Instance) => templates?.find((t) => t.id === i.template_id), [templates]);

  const all = useMemo(() => instances ?? [], [instances]);
  const rows = useMemo(() => {
    const q = query.trim().toLowerCase();
    return all
      .filter((i) => byFilter(i, filter))
      .filter((i) => !q || `${i.name} ${i.subdomain} ${tmpl(i)?.image ?? ""}`.toLowerCase().includes(q))
      .sort((a, b) => (ORDER[a.status] ?? 9) - (ORDER[b.status] ?? 9) || a.name.localeCompare(b.name));
  }, [all, query, filter, tmpl]);

  const selecting = picked.size > 0;
  const togglePick = (id: string) => setPicked((p) => {
    const n = new Set(p);
    if (n.has(id)) n.delete(id); else n.add(id);
    return n;
  });

  return (
    <section aria-label="Instances" className="mt-2">
      <GroupHeader title="Instances" count={all.length} collapsed={collapsed} onToggle={toggle} controlsId={LIST_ID}>
        <span className="flex items-center gap-2 text-xs text-muted-foreground max-sm:hidden">
          {COUNTS.map((c) => {
            const n = all.filter((i) => c.match(i.status)).length;
            return n ? (
              <span key={c.label} className="flex items-center gap-1" title={`${n} ${c.label}`}>
                <span className={cn("size-1.5 rounded-full", c.dot)} aria-hidden />
                <span className="tabular-nums text-foreground">{n}</span>
                <span className="sr-only">{c.label}</span>
              </span>
            ) : null;
          })}
        </span>
        <Button size="sm" variant="ghost" title="Filter by status"
          onClick={() => setFilter((f) => FILTERS[(FILTERS.indexOf(f) + 1) % FILTERS.length]!)}>
          <Filter aria-hidden />{FILTER_LABELS[filter]}
        </Button>
        <Button size="sm" variant={selecting ? "secondary" : "ghost"}
          onClick={() => setPicked(selecting ? new Set() : new Set(rows.map((i) => i.id)))}>
          <CheckSquare aria-hidden />{selecting ? `${picked.size} selected` : "Select"}
        </Button>
        <Button size="sm" onClick={onLaunch}><Plus aria-hidden />Launch</Button>
      </GroupHeader>
      {!collapsed && (
        <div id={LIST_ID} className="py-1">
          {isError ? (
            <div className="flex flex-wrap items-center gap-2 px-2 py-3 text-sm text-muted-foreground">
              <span>Backend unavailable — retrying…</span>
              <Button size="sm" variant="secondary" onClick={() => void refetch()}>Retry now</Button>
            </div>
          ) : !instances ? (
            <div className="h-12 animate-pulse rounded-lg bg-muted" />
          ) : all.length === 0 ? (
            <EmptyLine text="No instances yet." action={{ label: "Launch an instance", onClick: onLaunch }} />
          ) : rows.length === 0 ? (
            <p className="px-2 py-2 text-xs text-muted-foreground">No instances match.</p>
          ) : (
            <ul className="space-y-0.5">
              {rows.map((i) => {
                const m = statusMeta(i.status);
                return (
                  <WorkspaceRow
                    key={i.id}
                    icon={<TemplateIcon icon={tmpl(i)?.icon ?? null} />}
                    name={i.name}
                    line={instLine(i)}
                    dot={{ dotClass: cn(m.dotClass, m.pulse && "animate-pulse"), label: m.label }}
                    dim={m.group === "stopped" || m.group === "error"}
                    selected={selection?.kind === "instance" && selection.id === i.id}
                    onSelect={() => (selecting ? togglePick(i.id) : onSelect({ kind: "instance", id: i.id }))}
                    quick={isRunning(i.status) && !selecting ? { label: `Connect to ${i.name}`, onClick: () => connect(i) } : undefined}
                    leading={
                      <input type="checkbox" aria-label={`Select ${i.name}`} checked={picked.has(i.id)} onChange={() => togglePick(i.id)}
                        className={cn("size-4 shrink-0 accent-primary",
                          !selecting && "opacity-0 group-hover:opacity-100 focus-visible:opacity-100 pointer-coarse:hidden")} />
                    }
                  />
                );
              })}
            </ul>
          )}
        </div>
      )}
      <InstanceBulkBar selected={all.filter((i) => picked.has(i.id))} onClear={() => setPicked(new Set())} />
    </section>
  );
}
```

```tsx
import { AlertTriangle, ExternalLink } from "lucide-react";
import type { Workstation } from "@/api/client";
import { Button } from "@/components/ui/button";
import { WorkstationIcon } from "@/components/system/workstation-icon";
import { useWorkstationConnect } from "@/hooks/use-workstation-connect";
import { since, wsDot } from "@/lib/workspace";
import { PaneHeader } from "./pane-header";

function facts(ws: Workstation): [string, string][] {
  const os = ws.os_info as Record<string, unknown>;
  const gpu = ws.gpu_info as Record<string, unknown>;
  return [
    ["System", String(os.pretty_name || os.distro || "Linux")],
    ["GPU", String(gpu.model || gpu.vendor || "None detected")],
    ["Display", ws.display_server || "—"],
    ["Agent", ws.agent_version ? `v${ws.agent_version}${ws.agent_outdated ? " · update available" : ""}` : "—"],
  ];
}

/** Rich workstation card (no image yet: the last-frame preview waits for the agent batch). */
export function WorkstationDetailPane({ ws, isAdmin }: { ws: Workstation; isAdmin: boolean }) {
  const { connect, takeOver, dialog, error } = useWorkstationConnect();
  const online = ws.status === "online";
  const other = online && ws.in_use && !ws.in_use_self;
  const seen = since(ws.last_heartbeat);
  const dot = wsDot(ws);

  const actions = (
    <div className="flex flex-wrap gap-2">
      {online && (
        <Button variant="ghost" onClick={() => connect(ws, "new_tab")}><ExternalLink aria-hidden />Open in new tab</Button>
      )}
      {!online ? (
        <Button disabled>Offline</Button>
      ) : other ? (
        <>
          <Button onClick={() => connect(ws)}>Join</Button>
          {isAdmin && <Button variant="outline" onClick={() => takeOver(ws)}>Take over</Button>}
        </>
      ) : (
        <Button onClick={() => connect(ws)}>{ws.in_use_self ? "Reconnect" : "Connect"}</Button>
      )}
    </div>
  );

  return (
    <div className="flex h-full flex-col overflow-y-auto">
      <PaneHeader icon={<WorkstationIcon ws={ws} className="size-20 rounded-xl" />} title={ws.name}
        status={dot.label} dotClass={dot.dotClass} muted={!online} actions={actions} />
      <div className="space-y-4 p-4">
        {error && <p role="alert" className="text-sm text-destructive">{error}</p>}
        <dl className="grid grid-cols-2 gap-3 text-sm max-sm:grid-cols-1">
          {facts(ws).map(([k, v]) => (
            <div key={k} className="rounded-lg border border-border bg-card/50 p-3">
              <dt className="text-xs text-muted-foreground">{k}</dt>
              <dd className="truncate font-medium" title={v}>{v}</dd>
            </div>
          ))}
        </dl>
        {ws.last_error && (
          <div role="status" className="flex gap-2 rounded-md border border-warning/40 bg-warning/10 p-3 text-xs">
            <AlertTriangle className="mt-px size-4 shrink-0 text-warning" aria-hidden />
            <span>{ws.last_error}</span>
          </div>
        )}
        {(seen || !online) && (
          <p className="text-xs text-muted-foreground">
            {online ? `seen ${seen} ago` : seen ? `Offline · last seen ${seen} ago` : "Offline · never seen"}
          </p>
        )}
      </div>
      {dialog}
    </div>
  );
}
```
